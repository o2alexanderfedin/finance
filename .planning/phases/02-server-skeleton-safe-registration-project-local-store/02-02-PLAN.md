```
    randomNext: number
}
export declare const emptyState: State
export declare const virtual: RunInstance<NodeOp, State>
```
From `functionalscript/fjs/effects/mock/module.f.d.ts`:
```
export type RunInstance<O, S> = (state: S) => <O1 extends O, T>(effect: Effect<O1, T>) => readonly [S, T]
```
`virtual(state)(effect)` is **synchronous** — no `Promise`, no `awaitIfPromise` needed. A proof
leaf can call it directly and assert on the returned `[finalState, result]` tuple.

From `node_modules/functionalscript/fjs/protocol/mcp/stdio/proof.f.js` (byte-framing
precedent — read the whole file, not just this excerpt):
```
const toBytes = (s) => [...fromVec(utf8(s))]
const runStep = (step) => (input) => virtual({ ...emptyState, stdin: toBytes(input) })(stdioTransport(step))[0]
```
`fromVec`/`utf8` come from `functionalscript/fjs/types/uint8array/module.f.js` and
`functionalscript/fjs/text/module.f.js` respectively.

From `node_modules/functionalscript/fjs/protocol/mcp/proof.f.js` (message-shape precedent —
read the whole file): exact JSON-RPC request shapes for `initialize`
(`{ jsonrpc: '2.0', method: 'initialize', id, params: { protocolVersion, capabilities: {},
clientInfo: { name, version } } }`) and `notifications/initialized`
(`{ jsonrpc: '2.0', method: 'notifications/initialized' }` — no `id`, so `mcpStep` returns
`null` for it and the transport writes nothing).

From `node_modules/functionalscript/fjs/mcp/evo/module.f.js`: the `evo_list` tool takes `{}`
(no required args) and is always safe/read-only — use it as the `tools/call` target so this
proof needs no CAS writes.

From `functionalscript/fjs/asserts/module.f.js`: `assert(condition, msg?)`,
`assertEq(actual, expected, msg?)`.

Plain built-in `JSON.parse`/`JSON.stringify` are fine for building request strings and
decoding response lines in this proof — no need for `fjs/media/json`'s canonical-sort
`stringify`, since this proof is not comparing serialization byte-for-byte, only checking that
every stdout line parses as JSON-RPC and that specific fields hold the expected values.
